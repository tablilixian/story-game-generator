# 小说拆解功能集成开发方案

## 一、需求概述

### 1.1 功能目标
将小说文本自动拆解为结构化的章节、角色、场景数据，作为视觉小说游戏生成的输入起点。

### 1.2 核心功能
- **小说导入**：支持 TXT、PDF、EPUB 等格式的小说文本导入
- **智能分块**：基于段落和章节结构的智能文本分块
- **章节拆分**：自动识别和拆分小说章节结构
- **角色提取**：AI 分析并提取角色信息（名称、性别、年龄、性格）
- **场景提取**：AI 分析并提取场景信息（地点、时间、氛围）
- **预览编辑**：支持拆分结果预览和手动编辑

```typescript
/**
 * 小说文本分块
 * 基于段落和句子边界的智能分块
 */
export const chunkNovelText = (
  text: string,
  chunkSize: number = 3000,
  overlap: number = 500
): { chunks: string[]; positions: { start: number; end: number }[] } => {
  const paragraphs = text.split(/\n\n+/);
  const chunks: string[] = [];
  const positions: { start: number; end: number }[] = [];
  
  let currentChunk = '';
  let currentStart = 0;
  
  for (const paragraph of paragraphs) {
    if (currentChunk.length + paragraph.length > chunkSize && currentChunk.length > 0) {
      chunks.push(currentChunk.trim());
      positions.push({ start: currentStart, end: currentStart + currentChunk.length });
      
      // 处理重叠
      const overlapText = currentChunk.slice(-overlap);
      currentChunk = overlapText + '\n\n' + paragraph;
      currentStart = currentStart + currentChunk.length - overlap.length;
    } else {
      currentChunk += (currentChunk.length > 0 ? '\n\n' : '') + paragraph;
    }
  }
  
  if (currentChunk.trim().length > 0) {
    chunks.push(currentChunk.trim());
    positions.push({ start: currentStart, end: currentStart + currentChunk.length });
  }
  
  return { chunks, positions };
};
```

```typescript
// services/ai/chapterService.ts

import {
  retryOperation,
  chatCompletion,
  getActiveChatModel,
  getDefaultChatModelId,
} from './apiCore';
import { logger, LogCategory } from '../logger';

/**
 * 识别章节标题和编号
 */
const CHAPTER_PATTERNS = [
  /^(第[一二三四五六七八九十百千零\d]+[章节卷部篇集])\s*[\.、:：]\s*(.+)/,
  /^(Chapter|Chapter\s+\d+)\s*[\.、:：]\s*(.+)/i,
  /^(第[\d零一二三四五六七八九十]+)\s*[\.、:：]\s*(.+)/,
  /^\d+[\.、]\s*(.+)/,
  /^【(.+?)】\s*(.*)/,
];

/**
 * 拆分小说为章节
 */
export const splitIntoChapters = async (
  text: string,
  model?: string
): Promise<{ chapters: { title: string; content: string; order: number }[] }> => {
  const resolvedModel = model || getDefaultChatModelId();
  
  // 首先尝试正则匹配
  const lines = text.split('\n');
  const chapters: { title: string; content: string; order: number }[] = [];
  let currentChapter: { title: string; content: string[] } | null = null;
  let chapterOrder = 1;
  
  for (const line of lines) {
    const trimmedLine = line.trim();
    let matched = false;
    
    for (const pattern of CHAPTER_PATTERNS) {
      const match = trimmedLine.match(pattern);
      if (match) {
        if (currentChapter && currentChapter.content.length > 0) {
          chapters.push({
            title: currentChapter.title,
            content: currentChapter.content.join('\n'),
            order: chapterOrder++
          });
        }
        currentChapter = {
          title: match[1] + (match[2] ? ' ' + match[2] : ''),
          content: []
        };
        matched = true;
        break;
      }
    }
    
    if (!matched && currentChapter) {
      currentChapter.content.push(line);
    } else if (!matched && !currentChapter) {
      // 第一章之前的序言部分
      if (trimmedLine.length > 100) {
        currentChapter = {
          title: '序言',
          content: [line]
        };
      }
    }
  }
  
  // 处理最后一章
  if (currentChapter && currentChapter.content.length > 0) {
    chapters.push({
      title: currentChapter.title,
      content: currentChapter.content.join('\n'),
      order: chapterOrder
    });
  }
  
  // 如果正则匹配失败（章节格式不规整），使用 AI 辅助拆分
  if (chapters.length < 2) {
    logger.debug(LogCategory.AI, '正则匹配章节数不足，使用 AI 辅助拆分');
    return await aiAssistedChapterSplit(text, resolvedModel);
  }
  
  return { chapters };
};

/**
 * AI 辅助章节拆分（当正则匹配失败时使用）
 */
const aiAssistedChapterSplit = async (
  text: string,
  model: string
): Promise<{ chapters: { title: string; content: string; order: number }[] }> => {
  const prompt = `
    将以下小说文本拆分为章节，返回 JSON 格式。
    
    要求：
    1. 章节标题使用 "第X章 标题" 格式
    2. 每个章节内容完整
    3. 识别主要章节，可跳过子章节
    4. 返回纯 JSON，不要有其他文字
    
    文本：
    "${text.slice(0, 25000)}"
    
    输出格式：
    {
      "chapters": [
        {
          "title": "第一章 标题",
          "content": "章节内容",
          "order": 1
        }
      ]
    }
  `;
  
  const response = await retryOperation(() => 
    chatCompletion(prompt, model, 0.3, 8192, 'json_object')
  );
  
  try {
    const parsed = JSON.parse(response);
    return parsed;
  } catch {
    return { chapters: [{ title: '全文', content: text, order: 1 }] };
  }
};

/**
 * 生成章节摘要
 */
export const generateChapterSummary = async (
  chapterText: string,
  model?: string
): Promise<string> => {
  const resolvedModel = model || getDefaultChatModelId();
  
  const prompt = `
    为以下章节生成一段简短摘要（100字以内）：
    
    "${chapterText.slice(0, 5000)}"
    
    直接输出摘要，不要其他内容。
  `;
  
  return await retryOperation(() => 
    chatCompletion(prompt, resolvedModel, 0.3, 500)
  );
};
```

```typescript
/**
 * 解析 TXT 文件
 */
export const parseTxtFile = (text: string): { title: string; content: string } => {
  const lines = text.split('\n');
  let title = '未命名小说';
  let contentStartIndex = 0;
  
  // 尝试从开头几行提取标题
  for (let i = 0; i < Math.min(5, lines.length); i++) {
    const line = lines[i].trim();
    if (line.length > 0 && line.length < 100) {
      title = line;
      contentStartIndex = i + 1;
      break;
    }
  }
  
  const content = lines.slice(contentStartIndex).join('\n');
  return { title, content };
};

/**
 * 解析小说文件（统一入口）
 */
export const parseNovelFile = async (file: File): Promise<{ title: string; content: string }> => {
  const extension = file.name.split('.').pop()?.toLowerCase();
  
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    
    reader.onload = (e) => {
      const text = e.target?.result as string;
      
      switch (extension) {
        case 'txt':
          resolve(parseTxtFile(text));
          break;
        case 'md':
          resolve(parseTxtFile(text));
          break;
        case 'pdf':
          // PDF 解析需要引入 pdf.js
          reject(new Error('PDF 格式暂不支持，请转换为 TXT 格式'));
          break;
        case 'epub':
          // EPUB 解析需要引入 epub.js
          reject(new Error('EPUB 格式暂不支持，请转换为 TXT 格式'));
          break;
        default:
          reject(new Error(`不支持的文件格式: ${extension}`));
      }
    };
    
    reader.onerror = () => reject(new Error('文件读取失败'));
    reader.readAsText(file);
  });
};
```

### 5.3 文本分块服务

```typescript
// services/novel/textChunker.ts

export interface TextChunk {
  id: string;
  text: string;
  startIndex: number;
  endIndex: number;
  wordCount: number;
}

/**
 * 基于段落边界的智能分块
 */
export const createParagraphAwareChunks = (
  text: string,
  maxChunkSize: number = 3000,
  overlapSize: number = 500
): TextChunk[] => {
  const paragraphs = text.split(/\n\n+/);
  const chunks: TextChunk[] = [];
  
  let currentChunk = '';
  let currentStart = 0;
  let chunkId = 0;
  
  for (const paragraph of paragraphs) {
    const trimmedParagraph = paragraph.trim();
    
    if (trimmedParagraph.length === 0) continue;
    
    // 如果单个段落就超过了最大分块大小，需要进一步拆分
    if (trimmedParagraph.length > maxChunkSize) {
      if (currentChunk.length > 0) {
        chunks.push(createChunk(chunkId++, currentChunk, currentStart, currentStart + currentChunk.length));
        currentChunk = '';
      }
      
      // 按句子拆分大段落
      const sentences = trimmedParagraph.split(/[。！？.!?]+/);
      let subChunk = '';
      
      for (const sentence of sentences) {
        if (subChunk.length + sentence.length > maxChunkSize && subChunk.length > 0) {
          chunks.push(createChunk(chunkId++, subChunk, currentStart, currentStart + subChunk.length));
          currentStart += subChunk.length;
          subChunk = '';
        }
        subChunk += sentence + '。';
      }
      
      if (subChunk.trim().length > 0) {
        currentChunk = subChunk;
        currentStart = currentStart + subChunk.length - subChunk.length; // 实际应该更新
      }
    } else if (currentChunk.length + trimmedParagraph.length + 2 > maxChunkSize) {
      // 当前块满了，保存并创建新块
      chunks.push(createChunk(chunkId++, currentChunk, currentStart, currentStart + currentChunk.length));
      
      // 使用末尾重叠
      const overlap = currentChunk.slice(-overlapSize);
      currentChunk = overlap + '\n\n' + trimmedParagraph;
      currentStart = currentStart + currentChunk.length - trimmedParagraph.length - overlap.length;
    } else {
      // 添加到当前块
      if (currentChunk.length > 0) {
        currentChunk += '\n\n';
      }
      currentChunk += trimmedParagraph;
    }
  }
  
  // 处理最后一个块
  if (currentChunk.trim().length > 0) {
    chunks.push(createChunk(chunkId, currentChunk, currentStart, currentStart + currentChunk.length));
  }
  
  return chunks;
};

const createChunk = (
  id: number,
  text: string,
  startIndex: number,
  endIndex: number
): TextChunk => ({
  id: `chunk-${id}`,
  text,
  startIndex,
  endIndex,
  wordCount: countChineseWords(text)
});

/**
 * 统计中文字数（简单估算）
 */
const countChineseWords = (text: string): number => {
  const chineseChars = text.match(/[\u4e00-\u9fa5]/g) || [];
  const englishWords = text.match(/[a-zA-Z]+/g) || [];
  return chineseChars.length + englishWords.length;
};
```

```tsx
  // 处理角色和场景提取
  const handleExtraction = useCallback(async () => {
    if (!novelData || chapters.length === 0) return;
    
    setIsProcessing(true);
    const totalSteps = chapters.length + 2;
    let currentStep = 0;
    
    try {
      // 提取角色（基于全文）
      setProgress({ current: ++currentStep, total: totalSteps, message: '正在提取角色...' });
      const extractedChars = await extractCharacters(novelData.rawText);
      setCharacters(extractedChars);
      
      // 提取场景（基于全文）
      setProgress({ current: ++currentStep, total: totalSteps, message: '正在提取场景...' });
      const extractedScenes = await extractScenes(novelData.rawText);
      setScenes(extractedScenes);
      
      // 更新章节中的角色和场景引用
      for (const chapter of chapters) {
        setProgress({ current: ++currentStep, total: totalSteps, message: `分析章节: ${chapter.title}` });
        
        // 简单关键词匹配
        const chapterChars = extractedChars.filter(c => 
          chapter.rawText.includes(c.name)
        ).map(c => c.name);
        
        const chapterScenes = extractedScenes.filter(s => 
          chapter.rawText.includes(s.name) || 
          chapter.rawText.includes(s.location || '')
        ).map(s => s.name);
        
        chapter.characters = [...new Set(chapterChars)];
        chapter.scenes = [...new Set(chapterScenes)];
        chapter.status = 'completed';
      }
      
      setChapters([...chapters]);
      setStep('preview');
      
      logger.info(LogCategory.AI, `数据提取完成: ${extractedChars.length} 个角色, ${extractedScenes.length} 个场景`);
    } catch (error: any) {
      showAlert(`数据提取失败: ${error.message}`, { type: 'error' });
      logger.error(LogCategory.AI, '数据提取失败:', error);
    } finally {
      setIsProcessing(false);
    }
  }, [novelData, chapters]);

  // 完成导入，传递数据到下一阶段
  const handleComplete = useCallback(() => {
    const analysis: NovelAnalysis = {
      novelId: novelData?.id || '',
      chapters,
      characters,
      scenes,
      status: 'completed',
      progress: {
        chaptersCompleted: chapters.length,
        chaptersTotal: chapters.length,
        charactersExtracted: characters.length,
        scenesExtracted: scenes.length
      }
    };
    
    updateProject({
      analysis,
      // 将提取的数据传递到 scriptData
      scriptData: {
        ...project.scriptData,
        title: novelData?.title || '未命名',
        characters: characters.map(c => ({
          id: c.id,
          name: c.name,
          gender: c.gender || '',
          age: c.age || '',
          personality: c.personality || '',
          variations: []
        })),
        scenes: scenes.map(s => ({
          id: s.id,
          location: s.location || s.name,
          time: s.time || '',
          atmosphere: s.atmosphere || ''
        }))
      }
    });
    
    onComplete();
  }, [novelData, chapters, characters, scenes, project, updateProject, onComplete]);

  return (
    <div className="h-full flex">
      {/* 左侧主区域 */}
      <div className="flex-1 p-6">
        {step === 'import' && (
          <NovelImportPanel
            onFileImport={handleFileImport}
            isProcessing={isProcessing}
          />
        )}
        
        {step === 'split' && (
          <ChapterList
            chapters={chapters}
            onConfirm={handleChapterSplit}
            onCancel={() => setStep('import')}
            isProcessing={isProcessing}
          />
        )}
        
        {step === 'extract' && (
          <ExtractedDataPanel
            characters={characters}
            scenes={scenes}
            onConfirm={handleExtraction}
            onSkip={handleExtraction}
            isProcessing={isProcessing}
          />
        )}
        
        {step === 'preview' && (
          <div className="space-y-6">
            <div className="bg-[var(--bg-primary)] rounded-xl p-6">
              <h3 className="text-lg font-medium mb-4">导入完成</h3>
              <div className="grid grid-cols-3 gap-4">
                <div className="bg-[var(--bg-secondary)] rounded-lg p-4">
                  <div className="text-2xl font-bold">{chapters.length}</div>
                  <div className="text-sm text-[var(--text-muted)]">章节数</div>
                </div>
                <div className="bg-[var(--bg-secondary)] rounded-lg p-4">
                  <div className="text-2xl font-bold">{characters.length}</div>
                  <div className="text-sm text-[var(--text-muted)]">角色数</div>
                </div>
                <div className="bg-[var(--bg-secondary)] rounded-lg p-4">
                  <div className="text-2xl font-bold">{scenes.length}</div>
                  <div className="text-sm text-[var(--text-muted)]">场景数</div>
                </div>
              </div>
              <button
                onClick={handleComplete}
                className="mt-6 w-full py-3 bg-[var(--accent)] text-white rounded-lg hover:opacity-90"
              >
                进入剧本编辑
              </button>
            </div>
          </div>
        )}
      </div>
      
      {/* 右侧进度面板 */}
      {isProcessing && (
        <ImportProgress
          current={progress.current}
          total={progress.total}
          message={progress.message}
        />
      )}
    </div>
  );
};

export default StageImport;
```

```tsx
// components/StageImport/NovelImportPanel.tsx

import React, { useRef, useCallback } from 'react';
import { Upload, FileText } from 'lucide-react';

interface Props {
  onFileImport: (file: File) => void;
  isProcessing: boolean;
}

const NovelImportPanel: React.FC<Props> = ({ onFileImport, isProcessing }) => {
  const fileInputRef = useRef<HTMLInputElement>(null);
  
  const handleDrop = useCallback((e: React.DragEvent) => {
    e.preventDefault();
    const file = e.dataTransfer.files[0];
    if (file) {
      const ext = file.name.split('.').pop()?.toLowerCase();
      if (['txt', 'md'].includes(ext || '')) {
        onFileImport(file);
      }
    }
  }, [onFileImport]);
  
  const handleFileSelect = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      onFileImport(file);
    }
  }, [onFileImport]);

  return (
    <div className="max-w-2xl mx-auto">
      <h2 className="text-2xl font-bold mb-2">导入小说</h2>
      <p className="text-[var(--text-muted)] mb-8">
        支持 TXT、Markdown 格式
      </p>
      
      <div
        className={`
          border-2 border-dashed rounded-xl p-12 text-center
          transition-colors cursor-pointer
          ${isProcessing 
            ? 'border-[var(--border-primary)] bg-[var(--bg-primary)]' 
            : 'border-[var(--accent)] hover:bg-[var(--accent-bg)]'
          }
        `}
        onDrop={handleDrop}
        onDragOver={(e) => e.preventDefault()}
        onClick={() => !isProcessing && fileInputRef.current?.click()}
      >
        <input
          ref={fileInputRef}
          type="file"
          accept=".txt,.md"
          onChange={handleFileSelect}
          className="hidden"
```
